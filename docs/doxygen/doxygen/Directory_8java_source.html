144 public void move(Path destinationPath) throws IOException {
145 if (Files.exists(destinationPath)) {
146 throw new IOException("Path at " + destinationPath + " is already in use!");
147 }
148 if (destinationPath.toAbsolutePath().startsWith(path().toAbsolutePath())) {
149 if (destinationPath.toAbsolutePath().equals(path().toAbsolutePath())) {
150 throw new IOException("Directory is already located in the given destination!");
151 }
152 throw new IOException("Cannot move a directory into one of its own children!");
153 }
154 Map<Path, ZeppelinFile> movedChildren = new HashMap<>();
155 for (ZeppelinFile child : children.values()) {
156 child = child.load();
157 ZeppelinFile movedChild = child.copy(destinationPath.resolve(child.path().getFileName()));
158 movedChildren.put(movedChild.path(), movedChild);
159 }
160
161 Directory movedDirectory = new Directory(destinationPath, movedChildren);
162 movedDirectory.save();
163 delete();
164 }

167 move(destinationDirectory.path.resolve(path().getFileName()));

177 public Directory copy(Path destinationPath) throws IOException {
178 if (Files.exists(destinationPath)) {
179 throw new FileAlreadyExistsException("Path at " + destinationPath + " is already in use!");
180 }
181 Map<Path, ZeppelinFile> copyChildren = new HashMap<>();
182 for (ZeppelinFile child : children.values()) {
183 child = child.load();
184 String childCopyFileName = child.path().getFileName().toString();
185 Path copyChildPath = destinationPath.resolve(childCopyFileName);
186 copyChildren.put(copyChildPath, child.copy(copyChildPath));
187 }
188 Directory copiedDirectory = new Directory(destinationPath, copyChildren);
189 copiedDirectory.save();
190 return copiedDirectory;
191 }

246 // This method traverses the file tree recursively and depth first, and creates a Directory object with a complete map of child Directories and Notebooks.
247 public Directory initializeDirectory(Path pathToVisit, Map<Path, ZeppelinFile> existingFiles) throws IOException {
248 // Create a copy of existingFiles so that we don't make any direct edits to it.
249 Map<Path, ZeppelinFile> directoryChildren = new HashMap<>();
250 directoryChildren.putAll(existingFiles);
251 Files.walkFileTree(pathToVisit, new SimpleFileVisitor<Path>() {
252
253 public FileVisitResult preVisitDirectory(Path dir, BasicFileAttributes attrs) throws IOException {
254 // walkFileTree visits the root directory it's called on.
255 // This method operates on the given directory's children so we skip the processing of the root directory here.
256 if (dir.equals(pathToVisit)) {
257 return FileVisitResult.CONTINUE;
258 }
259 // Ignore .git directory. This could be moved somewhere else.
260 if (dir.startsWith(pathToVisit + "/.git")) {
261 return FileVisitResult.SKIP_SUBTREE;
262 }
263 // Here we create any child Directories by calling this function recursively.
264 // First we will check if we already have the files of the child Directory in existingFiles, and pass them to the recursive call so that we don't do any unnecessary operations in later recursions.
265 ConcurrentHashMap<Path, ZeppelinFile> childrenOfChildDirectory = new ConcurrentHashMap<>();
266 if (directoryChildren.containsKey(dir)) {
267 childrenOfChildDirectory.putAll(directoryChildren.get(dir).children());
268 }
269 Directory childDirectory = initializeDirectory(dir, childrenOfChildDirectory);
270 directoryChildren.put(childDirectory.path(), childDirectory);
271 return FileVisitResult.SKIP_SUBTREE;
272 }
273
274 @Override
275 public FileVisitResult visitFile(Path file, BasicFileAttributes attrs) {
276 if (!directoryChildren.containsKey(file)) {
277 directoryChildren.put(file, new Notebook(file));
278 }
279 return FileVisitResult.CONTINUE;
280 }
281
282 @Override
283 public FileVisitResult visitFileFailed(Path file, IOException exc) throws IOException {
284 return super.visitFileFailed(file, exc);
285 }
286
287 @Override
288 public FileVisitResult postVisitDirectory(Path dir, IOException exc) throws IOException {
289 return super.postVisitDirectory(dir, exc);
290 }
291 });
292 Directory root = new Directory(pathToVisit, directoryChildren);
293 return root;
294 }

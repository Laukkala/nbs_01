67// Copies a Notebook. Should be provided with a path of the File and a path of the source notebook to be copied.
68public final class CopyNotebookEndpoint implements FileSystemEndPoint {
69
70 private final Directory root;
71
72 public CopyNotebookEndpoint(Directory root) {
73 this.root = root;
74 }
75
76 public Response createResponse(Request request) {
77 try {
78 JsonObject parameters = request.parameters();
79 String sourcePath = parameters.getString("sourcePath");
80
81 Directory updatedDirectory = root.initializeDirectory(root.path(), root.children());
82 Path path = updatedDirectory.path().resolve(request.path());
83
84 ZeppelinFile newFile = copyNotebook(updatedDirectory, updatedDirectory.path().resolve(sourcePath), path);
85 return createResponse(newFile, parameters);
86 }
87 catch (FileNotFoundException fileNotFoundException) {
88 return new ExceptionResponse(HttpStatus.NOT_FOUND_404, fileNotFoundException);
89 }
90 catch (FileAlreadyExistsException fileAlreadyExistsException) {
91 return new ExceptionResponse(HttpStatus.BAD_REQUEST_400, fileAlreadyExistsException);
92 }
93 catch (IOException ioException) {
94 return new ExceptionResponse(HttpStatus.INTERNAL_SERVER_ERROR_500, ioException);
95 }
96 catch (MalformedRequestException malformedRequestException) {
97 return new ExceptionResponse(HttpStatus.BAD_REQUEST_400, malformedRequestException);
98 }
99 }
100
101 private Directory createDirectory(Path path) {
102 return new Directory(path);
103 }
104
105 private Notebook createNotebook(String title, Path path) {
106 return new Notebook(title, path, new HashMap<>());
107 }
108
109 private Notebook copyNotebook(Directory sourceDir, Path sourcePath, Path destinationPath) throws IOException {
110 ZeppelinFile file = sourceDir.findFile(sourcePath).load();
111 if (!file.isDirectory()) {
112 return (Notebook) file.copy(destinationPath);
113 }
114 else {
115 throw new IOException("File at " + sourcePath + " is not a notebook!");
116 }
117 }
118
119 @Override
120 public Response createResponse(ZeppelinFile file, JsonObject parameters) {
121 try {
122 file.save();
123 return new JsonResponse(HttpStatus.CREATED_201, "Created new notebook " + file.path());
124 }
125 catch (IOException ioException) {
126 return new ExceptionResponse(HttpStatus.INTERNAL_SERVER_ERROR_500, ioException);
127 }
128 }
129

131 public Directory root() {

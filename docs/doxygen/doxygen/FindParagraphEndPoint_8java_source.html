66// Searches for a paragraph from a given Notebook based on a given ParagraphId, and returns its contents in JSON format.
67public final class FindParagraphEndPoint implements FileSystemEndPoint {
68
69 private final Directory root;
70
71 public FindParagraphEndPoint(Directory root) {
72 this.root = root;
73 }
74
75 public Response createResponse(Request request) {
76 // Find a notebooks from Directory structure based on given ID
77 try {
78 validateRequestParameters(request);
79 Path requestPath = request.path();
80 Path notebookPath = requestPath.subpath(0, requestPath.getNameCount() - 2);
81
82 String paragraphId = requestPath
83 .subpath(requestPath.getNameCount() - 1, requestPath.getNameCount())
84 .toString();
85 JsonObject parameters = Json
86 .createObjectBuilder(request.parameters())
87 .add("paragraphId", paragraphId)
88 .build();
89
90 Directory updatedDirectory = root.initializeDirectory(root.path(), root.children());
91 Path path = updatedDirectory.path().resolve(notebookPath);
92 ZeppelinFile file = updatedDirectory.findFile(path).load();
93 return createResponse(file, parameters);
94 }
95 catch (FileNotFoundException fileNotFoundException) {
96 return new ExceptionResponse(HttpStatus.NOT_FOUND_404, fileNotFoundException);
97 }
98 catch (IOException ioException) {
99 return new ExceptionResponse(HttpStatus.INTERNAL_SERVER_ERROR_500, ioException);
100 }
101 catch (MalformedRequestException malformedRequestException) {
102 return new ExceptionResponse(HttpStatus.BAD_REQUEST_400, malformedRequestException);
103 }
104 }
105
106 private void validateRequestParameters(Request request) throws MalformedRequestException {
107 Path requestPath = request.path();
108 if (requestPath.getNameCount() < 3) {
109 throw new MalformedRequestException(
110 "Request path must be in format \"{path/to/notebook}/paragraph/{paragraphId}\""
111 );
112 }
113 if (!requestPath.getName(requestPath.getNameCount() - 2).toString().equals("paragraph")) {
114 throw new MalformedRequestException(
115 "Request path must be in format \"{path/to/notebook}/paragraph/{paragraphId}\""
116 );
117 }
118 }
119
120 @Override
121 public Response createResponse(ZeppelinFile file, JsonObject parameters) {
122 try {
123 if (!file.isDirectory()) {
124 String paragraphId = parameters.getString("paragraphId");
125 Notebook notebook = (Notebook) file.load();
126 if (notebook.paragraphs().containsKey(paragraphId)) {
127 return new JsonResponse(
128 HttpStatus.OK_200,
129 notebook.paragraphs().get(paragraphId).json().toString()
130 );
131 }
132 else {
133 return new ExceptionResponse(
134 HttpStatus.BAD_REQUEST_400,
135 new MalformedRequestException("Paragraph not found!")
136 );
137 }
138 }
139 else {
140 return new ExceptionResponse(
141 HttpStatus.BAD_REQUEST_400,
142 new MalformedRequestException("Not a notebook")
143 );
144 }
145 }
146 catch (IOException ioException) {
147 return new ExceptionResponse(
148 HttpStatus.INTERNAL_SERVER_ERROR_500,
149 new MalformedRequestException("Failed to save notebook!")
150 );
151 }
152 }

147 @Override
148 public Notebook load() throws IOException {
149 String content = readFile().toString();
150 StringReader stringReader = new StringReader(content);
151 JsonReader jsonReader = Json.createReader(stringReader);
152 JsonObject object = jsonReader.readObject();
153 jsonReader.close();
154 stringReader.close();
155
156 String savedName = object.getString("name");
157 JsonArray paragraphJsonArray = object.getJsonArray("paragraphs");
158 Map<String, Paragraph> savedParagraphs = new LinkedHashMap<>();
159 for (JsonObject paragraphJson : paragraphJsonArray.getValuesAs(JsonObject.class)) {
160 Paragraph paragraph = new Paragraph().load(paragraphJson);
161 savedParagraphs.put(paragraph.id(), paragraph);
162 }
163 return new Notebook(savedName, path(), savedParagraphs);
164 }

171 public Notebook copy(String copyTitle, Path destinationPath) throws IOException {
172 if (Files.exists(destinationPath)) {
173 throw new FileAlreadyExistsException("Path at " + destinationPath + " is already in use!");
174 }
175 Map<String, Paragraph> copyParagraphs = new LinkedHashMap<String, Paragraph>();
176 for (Paragraph paragraph : paragraphs.values()) {
177 String copyParagraphId = UUID.randomUUID().toString();
178 Script copyScript = new Script(paragraph.script().text());
179 Paragraph copyParagraph = new Paragraph(copyParagraphId, paragraph.title(), copyScript);
180 copyParagraphs.put(copyParagraph.id(), copyParagraph);
181 }
182 Notebook copyNotebook = new Notebook(copyTitle, destinationPath, copyParagraphs);
183 copyNotebook.save();
184 return copyNotebook;
185 }

228 private String readFile() throws IOException {
229 List<String> lines = Files.readAllLines(path(), Charset.defaultCharset());
230 String concatenatedLines = lines.stream().map(n -> String.valueOf(n)).collect(Collectors.joining("\n"));
231 return concatenatedLines;
